import { Component, input } from '@angular/core';

/** A container whose role an expression decides, through the property binding. */
@Component({
  selector: 'pct-group',
  templateUrl: './group.html',
  host: {
    class: 'pct-group',
    '[role]': 'multiple() ? "listbox" : "radiogroup"',
    '[attr.aria-labelledby]': 'labelledBy()',
  },
})
export class PctGroup {
  protected readonly labelledBy = input<string>('');
  protected readonly multiple = input(false);
}
