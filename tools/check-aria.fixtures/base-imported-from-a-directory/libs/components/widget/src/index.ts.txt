import { Directive, input } from '@angular/core';

/**
 * The base `widget.ts` imports from `'.'`, this directory's `index.ts`, without the naming
 * inputs.
 */

  host: {
    '[attr.data-pct-size]': 'size()',
  },
})
export abstract class PctWidgetBase {
  protected readonly size = input<string>('md');
}
