<!-- The role sits here and not on the host, so the name has to be handed over. It is handed
     over more than once, because the element a trigger is follows the role it carries: one
     that can be typed into is an `<input>`, one that cannot is a `<button>`, and one with
     nothing of ours to draw is the platform's `<select>`, whose options are its own. Exactly one
     of them is ever in the tree, which is the question point 4 asks — and the answer it must
     accept. -->
<span class="pct-widget__box">
  @if (editable()) {
  <input
    class="pct-widget__trigger"
    type="text"
    role="combobox"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  />
  } @else if (native()) {
  <select
    class="pct-widget__trigger"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  >
    <option>Poland</option>
    <option>Anywhere</option>
  </select>
  } @else {
  <button
    class="pct-widget__trigger"
    type="button"
    role="combobox"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  >
    {{ label() }}
  </button>
  }
</span>

<!-- The panel a menu, and between it and an option of its own, in a map, an `<ng-container>`
     a directive stamps rows through, the name bound on it the directive's input. -->
@if (open()) {
<div
  class="pct-widget__panel"
  role="menu"
  [attr.aria-label]="ariaLabel() || null"
  [attr.aria-labelledby]="ariaLabelledby() || null"
>
  <svg class="pct-widget__map" viewBox="0 0 8 8">
    <ng-container [pctRows]="rows" [ariaLabel]="label()"><g role="option"><text>Elsewhere</text></g></ng-container>
  </svg>
</div>
}

<span class="pct-widget__error" data-pct-part="error" role="alert">{{ error() }}</span>
} @else if (hint()) {
<span class="pct-widget__hint" data-pct-part="hint">{{ hint() }}</span>
}
