import { Component, input } from '@angular/core';

/**
 * Chrome around somebody else's control: no role on the host and no widget of its own, so the
 * component is asked for no name — and a landmark the consumer may name all the same.
 */
/** The selector kept in a constant, out of the decorator's body. */
const selector = 'pct-chrome';

@Component({
  selector,
  templateUrl: './chrome.html',
  host: {
    class: 'pct-chrome',
  },
})
export class PctChrome {
  readonly ariaLabelledby = input<string>('');
  /** Which of the two spellings is standing — the template's `@if`, and nothing else. */
  protected readonly sectioned = input(true);
}
