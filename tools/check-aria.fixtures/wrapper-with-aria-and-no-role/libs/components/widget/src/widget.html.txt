<!-- The role sits here and not on the host, so the name has to be handed over. It is handed
     over TWICE, because the element a trigger is follows the role it carries: one that can be
     typed into is an `<input>`, one that cannot is a `<button>`. Exactly one of the two is
     ever in the tree, which is the question point 4 asks — and the answer it must accept. -->
<span class="pct-widget__box">
  @if (editable()) {
  <input
    class="pct-widget__trigger"
    type="text"
    role="combobox"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  />
  } @else {
  <button
    class="pct-widget__trigger"
    type="button"
    role="combobox"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  >
    {{ label() }}
  </button>
  }
</span>

<!-- The panel: a listbox owning its options directly and THROUGH a heading's group, which is
     the relation point 9 reads off the tree — the group is one of the two contexts an option
     may stand in, and nothing else may. -->
@if (open()) {
<div class="pct-widget__panel" role="listbox">
  <div aria-labelledby="pct-widget-nearby">
    <div class="pct-widget__heading" id="pct-widget-nearby">Nearby</div>
    <div role="option">Poland</div>
  </div>
  <div role="option">Anywhere</div>
</div>
}

<!-- One message line: the error takes it, the hint gives way (req-api-message). -->
@if (showError()) {
<span class="pct-widget__error" data-pct-part="error" role="alert">{{ error() }}</span>
} @else if (hint()) {
<span class="pct-widget__hint" data-pct-part="hint">{{ hint() }}</span>
}
