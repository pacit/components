import { Component, input, model, type InputSignal } from '@angular/core';

/** Decoration that offers a name: hidden from the tree, and asking for a string nobody reads. */
@Component({
  selector: 'pct-widget',
  templateUrl: './widget.html',
  host: {
    class: 'pct-widget',
    'aria-hidden': 'true',
  },
})
export class PctWidget {
  protected readonly ariaLabel = model.required<string>();
  readonly ariaLabelledby = (input<string>('')) as InputSignal<string>;
}
