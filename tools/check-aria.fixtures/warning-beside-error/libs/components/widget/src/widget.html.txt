<!-- The role sits here and not on the host, so the name has to be handed over. -->

  <button
    class="pct-widget__trigger"
    type="button"
    role="combobox"
    [attr.aria-label]="ariaLabel() || null"
    [attr.aria-labelledby]="ariaLabelledby() || null"
  >
    {{ label() }}
  </button>
</span>

<!-- One message line: the error takes it, the hint gives way — and the warning stands beside
     them in a conditional of its own, which is the defect. -->
@if (showError()) {
<span class="pct-widget__error" data-pct-part="error" role="alert">{{ error() }}</span>
} @else if (hint()) {
<span class="pct-widget__hint" data-pct-part="hint">{{ hint() }}</span>
}
@if (showWarning()) {
<span class="pct-widget__warning" data-pct-part="warning" role="status">{{ warning() }}</span>
}
