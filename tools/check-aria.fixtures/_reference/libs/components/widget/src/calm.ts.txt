import { Component, input } from '@angular/core';

/**
 * A message line under everything that hides nothing, the pass side of what points 6 and 7 refuse
 * (`HIDING`): its host written `aria-hidden="false"` and binding a class named `inert`. And under
 * nothing that sets a part's live region (`LIVE_PROPERTIES`): an `aria-relevant` left empty, and
 * a class named `aria-busy`.
 */
@Component({
  selector: 'pct-calm',
  templateUrl: './calm.html',
  host: {
    class: 'pct-calm',
    'aria-hidden': 'false',
    '[class.inert]': 'quiet()',
    'aria-relevant': '',
    '[class.aria-busy]': 'quiet()',
  },
})
export class PctCalm {
  protected readonly error = input<string>('');
  protected readonly warning = input<string>('');
  protected readonly hint = input<string>('');
  protected readonly quiet = input(false);
}
