import { Component } from '@angular/core';

 *

 * decorator in base.ts, which the pattern does read. The decorators read are held file by file:
 * a set of offsets shared across the files would take that one for this one, and the count
 * would agree with the pattern.
 *
 * The case carries its own copy of base.ts, so that the two offsets stay one when the
 * reference's base.ts gets changed.
 */ @Component({
  selector: 'pct-widget-zeta',
  template: '',
})
