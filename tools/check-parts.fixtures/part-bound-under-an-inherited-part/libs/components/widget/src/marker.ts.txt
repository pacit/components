import { Directive } from '@angular/core';
import { PctMarkerBase } from './base';

@Directive({
  selector: '[pctMarker]',
  host: {
    class: 'pct-marker',
    'data-pct-part': 'widget-marker',
  },
})
export class PctMarker extends PctMarkerBase {}

/**
 * A directive extending an exported one and writing no part: it carries `PctMarker`'s, merged
 * into it as the base's host block is merged into `PctMarker`. The reference passes only while
 * the gate reads inheritance that way — a part a class inherits on the class extending it, and a
 * part the class writes over a base that writes none.
 */
@Directive({
  selector: '[pctMarkerOutline]',
  host: {
    class: 'pct-marker-outline',
    '[attr.data-pct-part]': 'part()',
  },
})
export class PctMarkerOutline extends PctMarker {}

/**
 * A directive in the file with no part and no base, as `popover.ts` and `tooltip.ts` keep a
 * trigger beside the component. Point 1 reads each decorator's body: a pair counted over the
 * whole file would be one this class never wrote, and the reference would stop passing.
 */
@Directive({
  selector: '[pctMarkerSlot]',
  host: {
    class: 'pct-marker-slot',
  },
})
export class PctMarkerSlot {}
