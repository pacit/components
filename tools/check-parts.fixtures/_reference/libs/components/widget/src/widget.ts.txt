import {
  Component as Component,
  ElementRef,
  inject as injectRef,
} from '@angular/core';

/**
 * The path in backticks, a literal ngc reads as it reads a quoted one, and the library does not
 * write: the scanner reads it too.
 */
@Component({
  selector: 'pct-widget',
  templateUrl: `./widget.html`,
  styleUrl: './widget.scss',
})
export class PctWidget {
  /**
   * `inject` under a name of the file's own, which the library does not write: only `Component`
   * and `Directive` are decorators the pattern knows by name. `Component` is imported under an
   * alias that is its own name, which is no other name.
   */
  private readonly host = injectRef<ElementRef<HTMLElement>>(ElementRef);

  /**
   * The class's code names a part, as `chips.ts` and `select.base.ts` do in the library. It is
   * no `host` block, so the counter that holds one reads the decorator's body and not the file.
   */
  protected label(): HTMLElement | null {
    return this.host.nativeElement.querySelector('[data-pct-part="label"]');
  }
}

/**
 * A component with an empty template, as `number.ts` and `text.ts` keep one written into the
 * decorator: it brings no part, so point 1 lets it stand. Written in backticks, the other literal
 * ngc compiles a template from, which the library does not write.
 */
@Component({
  selector: 'pct-widget-spacer',
  template: ``,
})
export class PctWidgetSpacer {}
