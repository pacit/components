import { booleanAttribute, input, model, output } from '@angular/core';
import { FormValueControl, ValidationError } from '@angular/forms/signals';

/** A control that carries the block itself — the ordinary shape, seven of eight here. */
@Component({ selector: 'pct-widget', template: '' })
export class PctWidget implements FormValueControl<string> {
  /** The value — the contract's one required member. */
  readonly value = model('');

  /** Blocks the control and greys it. */
  readonly disabled = input(false, { transform: booleanAttribute });

  /** The form's verdict; shown only once `touched`. */
  readonly invalid = input(false, { transform: booleanAttribute });

  /** Whether the user has left the field once. */
  readonly touched = input(false, { transform: booleanAttribute });

  /** The form's validation errors. */
  readonly errors = input<readonly ValidationError.WithOptionalFieldTree[]>([]);

  /** Emitted on blur. */
  readonly touch = output<void>();

  /** A member outside the block, so the reading is shown not to demand a closed set. */
  readonly label = input<string>('');
}

/**
 * A control built per call, under the name of the file's own: the class in the function shadows
 * `PctWidget` there, and TypeScript compiles it. It has no `touched`. A pairing by name and file
 * takes it for the control above, which a header stands at.
 */
export function pctWidgetOf(initial: string) {
  @Component({ selector: 'pct-widget-of', template: '' })
  class PctWidget implements FormValueControl<string> {
    /** The value — the contract's one required member. */
    readonly value = model(initial);

    /** Blocks the control and greys it. */
    readonly disabled = input(false, { transform: booleanAttribute });
  }
  return PctWidget;
}
