import { booleanAttribute, input, model, output } from '@angular/core';
import * as signals from '@angular/forms/signals';

/**
 * A control whose `implements` type wraps the contract, named through a namespace import, with
 * no `touched`. `Readonly<…>` of the contract is a type the class satisfies, and TypeScript
 * compiles it.
 */
@Component({ selector: 'pct-dial', template: '' })
export class PctDial implements Readonly<signals.FormValueControl<number>> {
  /** The value — the contract's one required member. */
  readonly value = model(0);

  /** Blocks the control and greys it. */
  readonly disabled = input(false, { transform: booleanAttribute });

  /** The form's verdict. */
  readonly invalid = input(false, { transform: booleanAttribute });

  /** The form's validation errors. */
  readonly errors = input<
    readonly signals.ValidationError.WithOptionalFieldTree[]
  >([]);

  /** Emitted on blur. */
  readonly touch = output<void>();
}
