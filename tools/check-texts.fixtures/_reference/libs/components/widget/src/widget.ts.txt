import {
  Component,
  computed,
  inject,
  input,
  isDevMode,
  signal,
} from '@angular/core';
import { PCT_TEXTS } from '../../core/src/texts';

export type PctWidgetSize = 'sm' | 'md' | 'lg';

@Component({
  selector: 'pct-widget',
  templateUrl: './widget.html',
  host: {
    class: 'pct-widget',
    role: 'group',
    '[attr.data-pct-size]': 'size()',
  },
})
export class PctWidget {
  protected readonly texts = inject(PCT_TEXTS);

  readonly label = input<string>('');
  readonly size = input<PctWidgetSize>('md');
  readonly items = input<readonly string[]>([]);

  /** A library string read AT RENDER TIME, not at construction. */
  protected readonly emptyText = computed(() => this.texts().widgetEmpty);
  protected readonly collapseText = computed(
    () => this.texts().widgetCollapse,
  );

  /**
   * A generic argument that is a FUNCTION type: the `>` of the arrow is not the one that
   * closes the type argument, and a scanner counting angle brackets alone stops there. Here
   * so that the shape is measured and not assumed.
   */
  protected readonly visible = computed<(item: string) => boolean>(
    () => (item) => item !== '',
  );

  protected readonly open = signal(true);

  toggle(): void {
    this.open.set(!this.open());
    if (isDevMode()) this.warnOnMisuse();
  }

  private warnOnMisuse(): void {
    if (!this.items().length) {
      console.warn('[pctWidget] Rendered without items.');
    }
  }
}
