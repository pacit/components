import { Directive, input } from '@angular/core';

/**
 * A slot directive — it exists so that the reference input carries a `host` block
 * with a static attribute whose key has NO quotes (`role`) beside a binding
 * whose key does. Prettier writes them exactly like that and the scanner must see
 * oba.
 */
  @Directive({
  selector: '[pctMarker]',
  host: {
    class: 'pct-marker',
    role: 'note',
    'data-pct-part': 'widget-marker',
    '[attr.aria-label]': 'description()',
  },
})
export class PctMarker {
  readonly description = input<string>('');
}
