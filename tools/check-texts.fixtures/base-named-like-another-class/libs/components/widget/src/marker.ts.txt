import { Directive, input, isDevMode } from '@angular/core';
import { PctMarkerBase } from './plain';

/**
 * A slot directive — it exists so that the reference input carries a `host` block
 * with a static attribute whose key has NO quotes (`role`) beside a binding
 * whose key does. Prettier writes them exactly like that and the scanner must see
 * both.
 *
 * Its base is the `PctMarkerBase` of `plain.ts`, which has no decorator, and not the
 * reference's decorated class of that name: the package carries the marker's own attributes
 * alone.
 */
@Directive({
  selector: '[pctMarker]',
  host: {
    class: 'pct-marker',
    'data-pct-part': 'widget-marker',
    title: '*',
    '[attr.aria-label]': 'description()',
  },
})
export class PctMarker extends PctMarkerBase {
  readonly description = input<string>('');
}

/**
 * A developer warning in a FREE FUNCTION rather than in a class method — the shape the
 * enclosure scanner could not see until D5. A free function's statements sit at the same
 * two-space indent a class puts its methods at, so `if (` read as the enclosing method and
 * the guard below was measured from the wrong place.
 */
export function pctWarnOnEmptyMarker(description: string): void {
  if (!isDevMode()) return;

  if (description === '') {
    console.warn('[pctMarker] Rendered with no description.');
  }
}
