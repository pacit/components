import { Directive, input } from '@angular/core';

/**
 * The reference base with one thing added: a SPEAKING attribute in its host block. The string
 * reaches the user only through the classes that extend it, which is where Angular puts it and
 * where point 2 has to find it; point 3 refuses it here, in the file that writes it.
 */
@Directive({
  host: {
    role: 'note',
    title: '',
    'aria-label': 'A note',
    '[attr.aria-description]': "count() ?? '–'",
  },
})
export abstract class PctMarkerBase {
  readonly size = input<'sm' | 'md' | 'lg'>('md');
  readonly count = input<number | null>(null);
}
