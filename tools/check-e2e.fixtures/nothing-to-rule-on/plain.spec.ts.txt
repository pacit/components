import { expect, test } from '@playwright/test';
import { visit } from './support/dom';

/** A spec with nothing for points 3 and 4 to rule on — no index, no number. */
test('the page names itself', async ({ page }) => {
  await visit(page, '/');
  await expect(page.getByRole('heading', { level: 1 })).toHaveText('Sandbox');
});
