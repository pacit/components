import { expect, test, type Locator } from '@playwright/test';
import { visit } from './support/dom';

/**
 * The hovered card's rim read through a function expression the spec defines: a helper as
 * plainly as an arrow is, and one the scanner has to read as one.
 */
const rimOf = function (locator: Locator) {
  return locator.evaluate((el) => getComputedStyle(el).borderColor);
};

test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await rimOf(page.locator('.card').first())).toBe('rgb(37, 99, 235)');
});
