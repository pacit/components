import { expect, test } from '@playwright/test';
import { styleOf } from './support/css';
import { visit } from './support/dom';

/**
 * The hovered card's rim read through a second name for `styleOf`, given in the spec: the
 * name is a call of the first, and reads as the first does.
 */
const colourOf = styleOf;

test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await colourOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
