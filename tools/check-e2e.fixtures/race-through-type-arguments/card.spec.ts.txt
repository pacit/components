import { expect, test, type Locator } from '@playwright/test';
import { visit } from './support/dom';

/**
 * Five reads of a card with type syntax around the reader: an arrow with type parameters,
 * one with a type parameter of its own inside them, a function declaration with them, one
 * whose return annotation holds an object type, and Playwright's own `evaluate<string>(…)`.
 */
const rimOf = <P extends string>(
  locator: Locator,
  property: P,
): Promise<string> =>
  locator.evaluate(
    (el, name) => getComputedStyle(el).getPropertyValue(name),
    property,
  );

const pick = <T extends Record<string, string>>(locator: Locator, _: T) =>
  locator.evaluate((el) => getComputedStyle(el).outlineColor);

async function frameOf<T extends Record<string, string>>(
  locator: Locator,
): Promise<{ colour: string; width: string }> {
  return locator.evaluate((el) => ({
    colour: getComputedStyle(el).borderColor,
    width: getComputedStyle(el).borderWidth,
  }));
}

function widthOf<P extends string>(locator: Locator, property: P) {
  return locator.evaluate(
    (el, name) => getComputedStyle(el).getPropertyValue(name),
    property,
  );
}

test('the first card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(
    await rimOf<'border-color'>(page.locator('.card').first(), 'border-color'),
  ).toBe('rgb(37, 99, 235)');
});

test('the second card thickens its rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').nth(1).hover();
  expect(
    await widthOf<'border-width'>(page.locator('.card').nth(1), 'border-width'),
  ).toBe('2px');
});

test('the last card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').last().hover();
  expect(
    await page
      .locator('.card')
      .last()
      .evaluate<string>((el) => getComputedStyle(el).borderColor),
  ).toBe('rgb(37, 99, 235)');
});

test('the focused card takes the accent outline', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').nth(2).focus();
  expect(
    await pick<Record<string, string>>(page.locator('.card').nth(2), {}),
  ).toBe('rgb(37, 99, 235)');
});

test('the last card frames itself in the accent', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').last().hover();
  expect((await frameOf(page.locator('.card').last())).colour).toBe(
    'rgb(37, 99, 235)',
  );
});
