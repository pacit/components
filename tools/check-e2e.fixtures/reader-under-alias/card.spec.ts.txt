import { expect, test } from '@playwright/test';
import { styleOf as colourOf } from './support/css';
import { visit } from './support/dom';

/**
 * The hovered card's race with its reader under another name: `colourOf` is `styleOf`, and
 * nothing the corpus defines is called `colourOf`, so the read is a call of no known kind.
 */
test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await colourOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
