import { expect, test } from '@playwright/test';
import tint, { colourOf } from './support';
import { visit } from './support/dom';

/** The hovered cards' rims, read through the names the support tree's index gives `styleOf`. */
test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await colourOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});

test('the last card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').last().hover();
  expect(await tint(page.locator('.card').last(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
