import { expect, test } from '@playwright/test';
import { visit } from './support/dom';
import palette from './support/palette';

/** The first card's rim, read through a default export that holds `styleOf` as a member. */
test('the first card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(
    await palette.styleOf(page.locator('.card').first(), 'border-color'),
  ).toBe('rgb(37, 99, 235)');
});
