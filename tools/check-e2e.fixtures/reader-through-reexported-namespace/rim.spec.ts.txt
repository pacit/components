import { expect, test } from '@playwright/test';
import { css, visit } from './support';

/** The hovered card's rim, read through the namespace the support tree's index re-exports. */
test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await css.styleOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
