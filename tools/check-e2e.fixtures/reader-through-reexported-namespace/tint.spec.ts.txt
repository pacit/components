import { expect, test } from '@playwright/test';
import { visit } from './support/dom';
import { $paint } from './support/paint';

/** The second card's rim, read through an object that holds a reader as a member. */
test('the second card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').nth(1).hover();
  expect(
    await $paint.rimOf<'border-color'>(
      page.locator('.card').nth(1),
      'border-color',
    ),
  ).toBe('rgb(37, 99, 235)');
});
