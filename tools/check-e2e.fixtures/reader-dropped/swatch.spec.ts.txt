import { expect, test, type Locator } from '@playwright/test';
import { visit } from './support/dom';

/**
 * A `styleOf` of this spec's own, as `visual.spec.ts` and `shell.spec.ts` each have a
 * `stage`: a helper name is file-local, so it says nothing of the one the others import.
 */
const styleOf = (locator: Locator, property: string) =>
  locator.evaluate(
    (el, prop) => getComputedStyle(el).getPropertyValue(prop),
    property,
  );

test('the accent swatch is painted the accent', async ({ page }) => {
  await visit(page, '/swatch');
  const swatch = page.getByTestId('swatch-accent');
  await expect(swatch).toBeVisible();
  expect(await styleOf(swatch, 'background-color')).toBe('rgb(37, 99, 235)');
});
