import { expect, test } from '@playwright/test';
import { visit } from './support/dom';
import * as css from './support/css';

/**
 * The hovered card's race with its reader reached through a namespace: `css.styleOf` is
 * `styleOf`, and to the scanner it is a method of an object rather than a helper by name.
 */
test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  expect(await css.styleOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
