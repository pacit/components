import { expect, test, type Page } from '@playwright/test';
import { styleOf } from './support/css';

/**
 * A `visit` of this spec's own, which navigates and waits for nothing — where the support
 * tree's `visit` waits for the shell's marker. Within this spec the name is this one, as
 * TypeScript has it, and the read after it stands in the window the navigation opened.
 */
const visit = async (page: Page, path: string) => {
  await page.goto(path);
};

test('the first card is painted the accent', async ({ page }) => {
  await visit(page, '/');
  expect(await styleOf(page.locator('.card').first(), 'border-color')).toBe(
    'rgb(37, 99, 235)',
  );
});
