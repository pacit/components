import { visit } from './support/dom';
import { expect, suiteTest as test } from './support/fixtures';

/** The hovered card's rim, asserted by a matcher of the suite's own. */
test('the hovered card takes the accent rim', async ({ page }) => {
  await visit(page, '/');
  await page.locator('.card').first().hover();
  await expect(page.locator('.card').first()).toHaveRim('rgb(37, 99, 235)');
});
