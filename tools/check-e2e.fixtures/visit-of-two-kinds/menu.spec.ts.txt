import { expect, test, type Page } from '@playwright/test';
import { visit } from './support/nav';

/** The page is ready once its main region is attached. */
const ready = (page: Page) => page.locator('main').waitFor();

/**
 * A menu read straight after `visit` from `support/nav.ts`, which only navigates, and after
 * a `ready` the third test defines to act where the spec's own waits.
 */
test.describe('Menu — the first item', () => {
  test('is labelled', async ({ page }) => {
    await visit(page, '/menu');
    expect(await page.locator('[role="menuitem"]').first().textContent()).toBe(
      'Copy',
    );
  });

  test('stays labelled after the menu opens', async ({ page }) => {
    await page.goto('/menu');
    await ready(page);
    await expect(page.locator('main')).toBeVisible();
  });

  test('reads its label once the menu is open', async ({ page }) => {
    const ready = (page: Page) =>
      page.getByRole('button', { name: 'Open' }).click();
    await page.goto('/menu');
    await expect(page.locator('main')).toBeVisible();
    await ready(page);
    expect(await page.locator('[role="menuitem"]').first().textContent()).toBe(
      'Copy',
    );
  });
});
